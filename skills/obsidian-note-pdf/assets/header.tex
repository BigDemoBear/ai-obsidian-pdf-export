% ---- headings ----------------------------------------------------------
\titleformat{\section}[block]
  {\normalfont\sffamily\bfseries\fontsize{18}{24}\selectfont\color{sectionblue}}
  {}{0pt}{}
  [\vspace{2.5pt}{\color{rulegray}\titlerule[0.6pt]}]
\titlespacing*{\section}{0pt}{3.0ex plus 1.0ex minus 0.2ex}{1.5ex}

\titleformat{\subsection}[hang]
  {\normalfont\sffamily\bfseries\fontsize{13.5}{18}\selectfont\color{sectionblue}}
  {}{0pt}{}
\titlespacing*{\subsection}{0pt}{2.5ex plus 0.7ex minus 0.2ex}{0.85ex}

\titleformat{\subsubsection}[hang]
  {\normalfont\sffamily\bfseries\fontsize{11.5}{15}\selectfont\color{black}}
  {}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{2.0ex plus 0.5ex minus 0.2ex}{0.65ex}

% Do not leave a heading stranded at the foot of a page.
\AddToHook{cmd/section/before}{\Needspace{9\baselineskip}}
\AddToHook{cmd/subsection/before}{\Needspace{7\baselineskip}}
\AddToHook{cmd/subsubsection/before}{\Needspace{5\baselineskip}}
